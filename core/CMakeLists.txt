cmake_minimum_required(VERSION 3.20)
project(HeliosCore LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(helios_core STATIC src/Simulation.cpp)
target_include_directories(helios_core PUBLIC include)
add_executable(helios_core_tests tests/main.cpp)
target_link_libraries(helios_core_tests PRIVATE helios_core)
enable_testing()
add_test(NAME helios_core_tests COMMAND helios_core_tests)
